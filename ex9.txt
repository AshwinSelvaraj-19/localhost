<?php
if (isset($_POST["calc"])) {

    $cid = (int) $_POST["cid"];
    $cname = trim($_POST["cname"]);
    $units = (int) $_POST["units"];

    if ($units < 0) {
        die("Units consumed cannot be negative.");
    }

    // Calculate bill amount using slab rates
    if ($units <= 100) {
        $bill_amt = $units * 3.00;
    } elseif ($units <= 300) {
        $bill_amt = (100 * 3.00) + (($units - 100) * 4.50);
    } else {
        $bill_amt = (100 * 3.00)
                  + (200 * 4.50)
                  + (($units - 300) * 6.00);
    }

    // Connect to MySQL server
    $conn = new mysqli("localhost", "root", "");

    if ($conn->connect_error) {
        die("Database connection failed: " . $conn->connect_error);
    }

    // Create and select database
    $conn->query("CREATE DATABASE IF NOT EXISTS eb_db");
    $conn->select_db("eb_db");

    // Create billing table
    $table_sql = "
        CREATE TABLE IF NOT EXISTS eb_bills (
            cid INT,
            cname VARCHAR(40),
            units INT,
            amount DECIMAL(10,2)
        )
    ";

    if (!$conn->query($table_sql)) {
        die("Table creation failed: " . $conn->error);
    }

    // Insert billing record securely
    $stmt = $conn->prepare(
        "INSERT INTO eb_bills (cid, cname, units, amount)
         VALUES (?, ?, ?, ?)"
    );

    $stmt->bind_param("isid", $cid, $cname, $units, $bill_amt);
    $stmt->execute();
    $stmt->close();

    echo "<div class='bill'>";
    echo "<h3>Electricity Bill</h3>";
    echo "Consumer ID: " . htmlspecialchars($cid) . "<br>";
    echo "Consumer Name: " . htmlspecialchars($cname) . "<br>";
    echo "Units Consumed: " . htmlspecialchars($units) . "<br>";
    echo "<strong>Total Bill: ₹" . number_format($bill_amt, 2) . "</strong>";
    echo "</div>";

    // Display all stored bills
    $result = $conn->query("SELECT * FROM eb_bills");

    echo "<h3>Stored Billing Records</h3>";
    echo "<table>";
    echo "<tr>
            <th>Consumer ID</th>
            <th>Name</th>
            <th>Units</th>
            <th>Amount</th>
          </tr>";

    while ($row = $result->fetch_assoc()) {
        echo "<tr>";
        echo "<td>" . htmlspecialchars($row["cid"]) . "</td>";
        echo "<td>" . htmlspecialchars($row["cname"]) . "</td>";
        echo "<td>" . htmlspecialchars($row["units"]) . "</td>";
        echo "<td>₹" . htmlspecialchars($row["amount"]) . "</td>";
        echo "</tr>";
    }

    echo "</table>";

    $conn->close();
}
