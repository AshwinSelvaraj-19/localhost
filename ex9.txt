    Consumer Name:
    <input type="text" name="cname" required>
    <br><br>

    Units Consumed:
    <input type="number" name="units" min="0" required>
    <br><br>

    <input type="submit" name="calc" value="Generate Invoice">

<?php
$host = "localhost";
$user = "root";
$pass = "";
$db = "eb_db";

// Connect to MySQL server
$conn = new mysqli($host, $user, $pass);

if ($conn->connect_error) {
    die("DBMS Connection Failed: " . $conn->connect_error);
}

// Create database
if (!$conn->query("CREATE DATABASE IF NOT EXISTS eb_db")) {
    die("Database Creation Failed: " . $conn->error);
}

// Select database
$conn->select_db($db);

// Create billing table
$table_sql = "
    CREATE TABLE IF NOT EXISTS eb_bills (
        cid INT,
        cname VARCHAR(40),
        units INT,
        amount DECIMAL(10,2)
    )
";

if (!$conn->query($table_sql)) {
    die("Table Creation Failed: " . $conn->error);
}

// Process submitted form
if (isset($_POST["calc"])) {
    $cid = (int) $_POST["cid"];
    $cname = trim($_POST["cname"]);
    $units = (int) $_POST["units"];

    if ($cid <= 0 || $units < 0 || $cname == "") {
        die("Please enter valid consumer details.");
    }

    // Calculate bill amount
    if ($units <= 100) {
        $bill_amt = $units * 3.00;
    } elseif ($units <= 300) {
        $bill_amt = (100 * 3.00) + (($units - 100) * 4.50);
    } else {
        $bill_amt = (100 * 3.00)
                  + (200 * 4.50)
                  + (($units - 300) * 6.00);
    }

    // Insert bill using prepared statement
    $insert_sql = "
        INSERT INTO eb_bills (cid, cname, units, amount)
        VALUES (?, ?, ?, ?)
    ";

    $stmt = $conn->prepare($insert_sql);
    $stmt->bind_param("isid", $cid, $cname, $units, $bill_amt);
    $stmt->execute();
    $stmt->close();

    echo "<h4>Invoice generated and saved successfully for consumer "
       . htmlspecialchars($cid)
       . "!</h4>";
}

// Display historical records
$result = $conn->query("SELECT * FROM eb_bills");

if ($result && $result->num_rows > 0) {
    echo "
    <h3>Historical Accounts Statement</h3>

    <table border='1' cellpadding='4' cellspacing='0'>
        <tr>
            <th>CID</th>
            <th>Name</th>
            <th>Units</th>
            <th>Total Bill</th>
        </tr>
    ";

    while ($row = $result->fetch_assoc()) {
        echo "
        <tr>
            <td>" . htmlspecialchars($row["cid"]) . "</td>
            <td>" . htmlspecialchars($row["cname"]) . "</td>
            <td>" . htmlspecialchars($row["units"]) . "</td>
            <td>₹" . number_format($row["amount"], 2) . "</td>
        </tr>
        ";
    }

    echo "</table>";
}

$conn->close();
?>
