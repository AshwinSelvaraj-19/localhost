<?php
if (isset($_POST["apply"])) {

    $conn = new mysqli("localhost", "root", "");

    if ($conn->connect_error) {
        die("Database Connection Failed: " . $conn->connect_error);
    }

    // Create database
    if (!$conn->query("CREATE DATABASE IF NOT EXISTS admission_db")) {
        die("Database Creation Failed: " . $conn->error);
    }

    // Select database
    $conn->select_db("admission_db");

    // Create table
    $table_sql = "
        CREATE TABLE IF NOT EXISTS applications (
            id INT AUTO_INCREMENT PRIMARY KEY,
            name VARCHAR(50) NOT NULL,
            phone VARCHAR(15) NOT NULL,
            stream VARCHAR(30) NOT NULL,
            aggregate DECIMAL(5,2) NOT NULL
        )
    ";

    if (!$conn->query($table_sql)) {
        die("Table Creation Failed: " . $conn->error);
    }

    // Get form values
    $name = trim($_POST["sname"]);
    $phone = trim($_POST["phone"]);
    $stream = $_POST["stream"];
    $marks = (float) $_POST["marks"];

    // Validate input
    if ($name == "" || $phone == "" || $marks < 0 || $marks > 100) {
        die("Please enter valid admission details.");
    }

    // Insert application using prepared statement
    $insert_sql = "
        INSERT INTO applications (name, phone, stream, aggregate)
        VALUES (?, ?, ?, ?)
    ";

    $stmt = $conn->prepare($insert_sql);
    $stmt->bind_param("sssd", $name, $phone, $stream, $marks);

    if ($stmt->execute()) {
        echo "<br><strong>Registration Successful!</strong>";
    } else {
        echo "<br>Registration Failed: " . $stmt->error;
    }

    $stmt->close();
    $conn->close();
}
?>
