<?php

if (isset($_POST['apply'])) {

    $name = $_POST['sname'];
    $phone = $_POST['phone'];
    $stream = $_POST['stream'];
    $marks = (float) $_POST['marks'];

    // Connect to MySQL server
    $conn = new mysqli("localhost", "root", "");

    if ($conn->connect_error) {
        die("Connection failed: " . $conn->connect_error);
    }

    // Create and select database
    $conn->query("CREATE DATABASE IF NOT EXISTS admissions_db");
    $conn->select_db("admissions_db");

    // Create applicants table
    $conn->query("
        CREATE TABLE IF NOT EXISTS applicants (
            id INT AUTO_INCREMENT PRIMARY KEY,
            name VARCHAR(50),
            phone VARCHAR(15),
            stream VARCHAR(30),
            aggregate FLOAT
        )
    ");

    // Prepared statement
    $stmt = $conn->prepare("
        INSERT INTO applicants (name, phone, stream, aggregate)
        VALUES (?, ?, ?, ?)
    ");

    // s = string, d = double/float
    $stmt->bind_param("sssd", $name, $phone, $stream, $marks);

    if ($stmt->execute()) {
        echo "<p class='success'>";
        echo "Registration Complete!<br>";
        echo "Application ID Assigned: " . $stmt->insert_id;
        echo "</p>";
    } else {
        echo "<p class='error'>Registration Error Encountered.</p>";
    }

    $stmt->close();
    $conn->close();
}
