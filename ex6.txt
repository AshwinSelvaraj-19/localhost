<?php
$marks = array(78, 92, 45, 88, 61, 99);

echo "<h3>Student Marks Array Sorter</h3>";

echo "Unsorted Marks: " . implode(", ", $marks) . "<br><br>";

// Sort in ascending order
sort($marks);

echo "Marks Sorted Ascending (Lowest First):<br>";

foreach ($marks as $m) {
    echo $m . " ";
}

// Sort in descending order
rsort($marks);

echo "<br><br>Marks Sorted Descending (Highest First):<br>";

foreach ($marks as $m) {
    echo $m . " ";
}
?>
