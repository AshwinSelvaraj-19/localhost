<?php

$filename = __DIR__ . "/lab_log.txt";

echo "<h3>File System Manipulation Engine</h3>";

$file_handle = fopen($filename, "w")
    or die("Unable to open the file for writing!");

$txt_payload = "Operating Systems and Web Scripting Lab Data.\n";
$txt_payload .= "Timestamp Log Registered Successfully.";

fwrite($file_handle, $txt_payload);
fclose($file_handle);

echo "Status: File created and text written successfully.<br>";

$file_handle = fopen($filename, "r")
    or die("Unable to open the file for reading!");

$file_content = fread($file_handle, filesize($filename));

fclose($file_handle);

echo "<br><strong>Content Extracted From File:</strong><br>";
echo nl2br(htmlspecialchars($file_content));

?>
