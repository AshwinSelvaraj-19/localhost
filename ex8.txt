<?php
$host = "localhost";
$user = "root";
$pass = "";
$db = "college_db";

// Connect to MySQL server
$conn = new mysqli($host, $user, $pass);

if ($conn->connect_error) {
    die("DBMS Connection Failed: " . $conn->connect_error);
}

// Create database
if (!$conn->query("CREATE DATABASE IF NOT EXISTS college_db")) {
    die("Database Creation Failed: " . $conn->error);
}

// Select database
$conn->select_db($db);

// Create table
$table_sql = "
    CREATE TABLE IF NOT EXISTS students (
        reg_no INT PRIMARY KEY,
        stud_name VARCHAR(50) NOT NULL,
        email VARCHAR(50) UNIQUE,
        course VARCHAR(30) DEFAULT 'B.Tech'
    )
";

if (!$conn->query($table_sql)) {
    die("Table Creation Failed: " . $conn->error);
}

// Clear previous records
if (!$conn->query("TRUNCATE TABLE students")) {
    die("Table Truncation Failed: " . $conn->error);
}

// Insert records using prepared statements
$insert_sql = "
    INSERT INTO students (reg_no, stud_name, email, course)
    VALUES (?, ?, ?, ?)
";

$stmt = $conn->prepare($insert_sql);

$reg_no = 101;
$stud_name = "Rahul Sharma";
$email = "rahul@email.com";
$course = "B.Tech";

$stmt->bind_param("isss", $reg_no, $stud_name, $email, $course);
$stmt->execute();

$reg_no = 102;
$stud_name = "Aditi Rao";
$email = "aditi@email.com";
$course = "M.Tech";

$stmt->execute();
$stmt->close();

// Update record
$update_sql = "
    UPDATE students
    SET course = ?
    WHERE reg_no = ?
";

$stmt = $conn->prepare($update_sql);

$updated_course = "B.Tech(Honors)";
$student_reg_no = 101;

$stmt->bind_param("si", $updated_course, $student_reg_no);
$stmt->execute();
$stmt->close();

// Select records
$result = $conn->query("SELECT * FROM students");

if (!$result) {
    die("Record Selection Failed: " . $conn->error);
}

// Display records
echo "<h3>MySQL Records Processing Matrix</h3>";

echo "
<table border='1' cellpadding='5' cellspacing='0'>
    <tr>
        <th>Reg No</th>
        <th>Name</th>
        <th>Email</th>
        <th>Course</th>
    </tr>
";

while ($row = $result->fetch_assoc()) {
    echo "
    <tr>
        <td>" . htmlspecialchars($row["reg_no"]) . "</td>
        <td>" . htmlspecialchars($row["stud_name"]) . "</td>
        <td>" . htmlspecialchars($row["email"]) . "</td>
        <td>" . htmlspecialchars($row["course"]) . "</td>
    </tr>
    ";
}

echo "</table>";

$conn->close();
?>
