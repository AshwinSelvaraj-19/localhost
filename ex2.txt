    <?php

        // Read form data

        $basic_sal = (float) $_POST['basic_sal'];

        // Salary calculations
        $hra = $basic_sal * 0.15;
        $da = $basic_sal * 0.20;
        $gross_salary = $basic_sal + $hra + $da;

        // Display result
        echo "<div class='result'>";
        echo "<h3>Employee Processing Slip</h3>";

        echo "Employee ID: " . htmlspecialchars($emp_id) . "<br>";
        echo "Employee Name: " . htmlspecialchars($emp_name) . "<br>";
        echo "Designation: " . htmlspecialchars($designation) . "<br>";
        echo "Basic Salary: ₹" . number_format($basic_sal, 2) . "<br>";
        echo "HRA (15%): ₹" . number_format($hra, 2) . "<br>";
        echo "DA (20%): ₹" . number_format($da, 2) . "<br>";

        echo "<strong>Gross Salary: ₹" .
             number_format($gross_salary, 2) . "</strong>";
